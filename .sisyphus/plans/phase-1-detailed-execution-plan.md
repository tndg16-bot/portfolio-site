### �量分析（探索時）

```bash
# タイトル長さの分析
for file in content/blog/*.md; do
  title=$(grep "^title:" "$file" | head -1 | cut -d'"' | sed 's/^"\(.*\)" *//')
  echo "$title: ${#title}"
done | awk '{sum+=$1} END {print "Average title length:", sum/NR}'

# カテゴリ分布確認
for file in content/blog/*.md; do
  category=$(grep "^category:" "$file" | head -1 | cut -d'"' | sed 's/^"\(.*\)" *//')
  echo "$category: ${#category}"
done | sort | uniq -c
```
